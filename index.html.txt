        // Page Navigation Logic (SPA Routing)
        function switchPage(pageId) {
            const pages = document.querySelectorAll('.page-view');
            pages.forEach(page => page.classList.add('hidden'));

            const navLinks = document.querySelectorAll('.nav-link');
            navLinks.forEach(link => link.classList.remove('active'));

            const targetPage = document.getElementById('page-' + pageId);
            if (targetPage) {
                targetPage.classList.remove('hidden');
            }

            const targetNav = document.getElementById('nav-' + pageId);
            if (targetNav) {
                targetNav.classList.add('active');
            }

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Generate Random Captcha
        function generateCaptcha() {
            const chars = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
            let captcha = "";
            for (let i = 0; i < 5; i++) {
                captcha += chars.charAt(Math.floor(Math.random() * chars.length));
            }
            document.getElementById('captcha-code').innerText = captcha;
        }

        // Registration Submission

            alert(`Account registered successfully for ${name}! Please login with your credentials.`);

        }

        // Login Submission
        function handleLoginSubmit(e) {
            e.preventDefault();
            const inputCaptcha = document.getElementById('login-captcha').value.toUpperCase();
            const currentCaptcha = document.getElementById('captcha-code').innerText;

            if (inputCaptcha !== currentCaptcha) {
                alert('Invalid Captcha Code. Please try again!');
                generateCaptcha();
                return;
            }

            const loginId = document.getElementById('login-id').value;
            document.getElementById('dash-user-name').innerText = `Welcome, ${loginId}`;
            document.getElementById('dashboard-modal').classList.remove('hidden');
        }

        // Live Chat Assistant Toggle & Messages

            document.getElementById('chat-box').classList.toggle('hidden');

        function sendChatMessage() {
            const input = document.getElementById('chat-input');
            const msg = input.value.trim();
            if (!msg) return;

            addMessage(msg, 'user');
            input.value = '';

            setTimeout(() => {
                let reply = "Thank you for contacting Mahadev Net Cafe Patiala! Call us at +91 81462 44491 for instant help.";
                const lower = msg.toLowerCase();
                if (lower.includes('pan') || lower.includes('fee')) {
                    reply = "New PAN Card and PAN Correction fee is ₹250/- with Aadhaar Card and Passport Photo.";
                } else if (lower.includes('register') || lower.includes('account')) {
                    reply = "You can click on 'Online Registration' in the top menu to create your citizen portal account.";
                } else if (lower.includes('location') || lower.includes('address')) {
                    reply = "We are located at Prem colony Near CIA Staff Patiala 147001.";
                }
                addMessage(reply, 'bot');
            }, 500);
        }

        function quickChat(type) {
            if (type === 'PAN Card Fees') {
                addMessage("What are the PAN Card fees?", 'user');
                setTimeout(() => addMessage("New PAN Card & PAN Correction fee is ₹250/-.", 'bot'), 400);
            } else if (type === 'Registration') {
                addMessage("How to register for account?", 'user');
                setTimeout(() => addMessage("Click on 'Online Registration' in the header menu to create your account.", 'bot'), 400);
            } else if (type === 'Address & Location') {
                addMessage("Where is your shop located?", 'user');
                setTimeout(() => addMessage("Prem colony Near CIA Staff Patiala 147001.", 'bot'), 400);
            }
        }

        function addMessage(text, sender) {
            const container = document.getElementById('chat-messages');
            const msgDiv = document.createElement('div');
            msgDiv.className = sender === 'user' 
                ? 'bg-brand-600 text-white p-3 rounded-xl ml-auto max-w-[85%] text-xs' 
                : 'bg-white p-3 rounded-xl border border-slate-200 max-w-[85%] text-xs text-slate-800';
            msgDiv.innerText = text;
            container.appendChild(msgDiv);
            container.scrollTop = container.scrollHeight;
        }
